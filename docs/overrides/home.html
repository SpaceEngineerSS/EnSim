EnSim

Liquid rocket engines from propellant to trajectory: chamber equilibrium, nozzle expansion, cooling and acoustics, then the vehicle flown in six degrees of freedom. A desktop application and a Python library.

Ten lines

from ensim.core.engine import EngineInputs, analyze_engine

engine = analyze_engine(EngineInputs(
    fuel="RP-1", oxidizer="LOX", of_ratio=2.6,
    chamber_pressure=70e5, expansion_ratio=25, throat_area=0.01,
))
print(f"Tc {engine.chamber_temperature:.0f} K")
print(f"Isp vacuum {engine.vacuum.isp:.1f} s, sea level {engine.sea_level.isp:.1f} s")
print(f"thrust {engine.vacuum.thrust / 1e3:.1f} kN")

# Tc 3673 K
# Isp vacuum 319.6 s, sea level 255.0 s
# thrust 125.3 kN
